--========================================================
-- Running Stock Balance
--
-- Calculate the inventory balance for each item at every
-- site by adding opening stock and restocks while
-- subtracting daily consumption.
--========================================================

--combined all inventory movements (restock and consumption) into a single timeline for each item at each site.

WITH Unified_timeline AS(
--Posive movement (restock) 
    SELECT
        site_id,
        item_id,
        date AS movement_date,
        qty AS movement_qty
    FROM inventory_management_analytics.restock

    UNION ALL

--negative movement (consumption)
    SELECT
        site_id,
        item_id,
        date AS movement_date,
        -qty_drawn AS movement_qty
    FROM inventory_management_analytics.daily_usage
    WHERE is_outlier = 'FALSE'
)
SELECT
    t.site_id,
    t.item_id,
    t.movement_date,
    o.opening_stock AS opening_stock,
    t.movement_qty AS movement_qty,
    ( COALESCE(o.opening_stock, 0)
     + SUM(t.movement_qty) OVER (PARTITION BY t.site_id, t.item_id 
     ORDER BY t.movement_date ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) ) AS balance_stock
FROM Unified_timeline t
LEFT JOIN inventory_management_analytics.opening_stock o
    ON t.site_id = o.site_id AND t.item_id = o.item_id
WHERE t.item_id = 'ITM-03' AND t.site_id = 'North_01'
ORDER BY t.movement_date

LIMIT 30;
